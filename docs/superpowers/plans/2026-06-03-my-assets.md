# My Assets Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a logged-in-only `/my-assets` user page for managing manually saved cloud image/video assets with search, filter, sort, preview, download, delete, and chat reuse.

**Architecture:** Reuse the existing saved-media persistence and save flow, add a dedicated user page and feature UI for asset management, and expose a server-controlled asset access endpoint for preview/download so the frontend never authorizes against raw storage fields. Keep route files thin and place derived UI logic in feature modules.

**Tech Stack:** Next.js App Router, React 19, TypeScript, zod, existing auth guards, existing user media repository/service, shadcn/Radix UI patterns already in repo

import { NextResponse } from 'next/server';
import { z } from 'zod';

const accessQuerySchema = z.object({
  disposition: z.enum(['preview', 'download']).default('preview'),
});

export function createMediaAssetAccessRouteHandlers(dependencies: {
  requireSession: () => Promise<{ user: { id: string } }>;
  getSavedAsset: (assetId: string, userId: string) => Promise<unknown>;
  createAccessUrl: (input: { assetId: string; disposition: 'preview' | 'download' }) => Promise<unknown>;
}) {
  return {
    async GET(request: Request, context: { params: Promise<{ assetId: string }> }) {
      const session = await dependencies.requireSession();
      const { assetId } = await context.params;
      const parsed = accessQuerySchema.safeParse(Object.fromEntries(new URL(request.url).searchParams));

      if (!parsed.success) {
        return NextResponse.json(
          { error: { code: 'invalid_request', message: parsed.error.issues[0]?.message ?? 'Invalid request.' } },
          { status: 400 },
        );
      }

      const asset = await dependencies.getSavedAsset(assetId, session.user.id);
      if (!asset) {
        return NextResponse.json(
          { error: { code: 'asset_not_found', message: 'Saved media asset was not found.' } },
          { status: 404 },
        );
      }

      const access = await dependencies.createAccessUrl({
        assetId,
        disposition: parsed.data.disposition,
      });

      return NextResponse.json({ access });
    },
  };
}
```

import type { GeneratedMediaAssetDto } from '@/server/agent/types';

export function createSavedMediaAccessService(dependencies: {
  signObjectUrl: (input: {
    objectKey: string;
    disposition: 'preview' | 'download';
    mimeType: string | null;
    title: string;
  }) => Promise<{ url: string; expiresAt: string }>;
}) {
  return {
    async createAccessUrl(input: {
      asset: Pick<GeneratedMediaAssetDto, 'id' | 'objectKey' | 'mimeType' | 'title'>;
      disposition: 'preview' | 'download';
    }) {
      const signed = await dependencies.signObjectUrl({
        objectKey: input.asset.objectKey,
        disposition: input.disposition,
        mimeType: input.asset.mimeType,
        title: input.asset.title,
      });

      return {
        assetId: input.asset.id,
        disposition: input.disposition,
        url: signed.url,
        expiresAt: signed.expiresAt,
        mimeType: input.asset.mimeType,
      };
    },
  };
}

import type { GeneratedMediaAssetDto } from '@/server/agent/types';

export type MyAssetsFilter = {
  search: string;
  kind: 'all' | 'image' | 'video';
  sort: 'newest' | 'oldest';
};

export function deriveMyAssetsView(
  assets: GeneratedMediaAssetDto[],
  filter: MyAssetsFilter,
): GeneratedMediaAssetDto[] {
  const normalizedSearch = filter.search.trim().toLowerCase();

  return assets
    .filter((asset) => (filter.kind === 'all' ? true : asset.kind === filter.kind))
    .filter((asset) =>
      normalizedSearch ? asset.title.toLowerCase().includes(normalizedSearch) : true,
    )
    .sort((left, right) =>
      filter.sort === 'oldest'
        ? left.savedAt.localeCompare(right.savedAt)
        : right.savedAt.localeCompare(left.savedAt),
    );
}
```
